    98→local function topOf(d)
    99→	local cf, s = d.CFrame, d.Size
   100→	return cf.Y + 0.5 * (math.abs(cf.RightVector.Y) * s.X + math.abs(cf.UpVector.Y) * s.Y + math.abs(cf.LookVector.Y) * s.Z)
   101→end
   102→local function solid(d)

   156→local function drawPile()
   157→	local want = math.min(math.ceil(pending / C.BUNDLE_VALUE), 30)
   158→	local have = #pile:GetChildren()
   159→	for i = have + 1, want do
   160→		local b = PROPS.CashBundle:Clone()
   161→		local k = i - 1
   162→		local layer = k // 6
   163→		local slot = k % 6
   164→		local x = (slot % 2) * 1.7 - 0.85
   165→		local z = (slot // 2) * 0.82 - 0.82
   166→		b:PivotTo(pileOrigin * CFrame.new(x, layer * 0.31, z) * CFrame.Angles(0, rng:NextNumber(-0.06, 0.06), 0))
   167→		for _, p in ipairs(b:GetDescendants()) do if p:IsA("BasePart") then p.CanCollide = false end end
   168→		b.Parent = pile
   169→	end
   170→	for i = have, want + 1, -1 do
   171→		local c = pile:GetChildren()[i]
   172→		if c then c:Destroy() end
   173→	end
   174→	if cashPrompt then
   175→		cashPrompt.Enabled = pending > 0
   176→		cashPrompt.ActionText = "Collect $" .. pending
   177→	end
   178→	HOTEL:SetAttribute("Pending", pending)
   179→end

   309→local function takeTip(room, player)
   310→	local t = room.tip
   311→	if not t then return end
   312→	room.tip = nil
   313→	local pos = t.part.Position
   314→	t.part:Destroy()
   315→	if player then
   316→		addCash(player, t.amount)
   317→	else
   318→		pending += t.amount
   319→		dayStats.earned += t.amount
   320→		drawPile()
   321→	end
   322→	sparkleBurst(pos)
   323→	Notify:FireAllClients("paid", { pos = pos + Vector3.new(0, 2, 0), amount = t.amount })
   324→end

   417→-- inner wall faces round a room (raycasts into the shell, so wing rooms work too)
   418→local function roomBox(room)
   419→	if room.box then return room.box end
   420→	local f = room.model.Floor
   421→	if room.model:GetAttribute("Custom") then
   422→		local ox = f.Position.X
   423→		room.box = { cx = ox, xl = ox - f.Size.X / 2, xr = ox + f.Size.X / 2, zb = f.Position.Z - f.Size.Z / 2, zf = f.Position.Z + f.Size.Z / 2 }
   424→		return room.box
   425→	end
   426→	local cx = f.Position.X
   427→	local rp = RaycastParams.new()
   428→	rp.FilterType = Enum.RaycastFilterType.Include
   429→	rp.FilterDescendantsInstances = { HOTEL.Shell }
   430→	local o = Vector3.new(cx, Build.FY + 6, -15)
   431→	local function hit(d, fallback)
   432→		local r = workspace:Raycast(o, d, rp)
   433→		return r and r.Position or fallback
   434→	end
   435→	room.box = {
   436→		cx = cx,
   437→		xl = hit(Vector3.new(-40, 0, 0), o - Vector3.new(12.75, 0, 0)).X,
   438→		xr = hit(Vector3.new(40, 0, 0), o + Vector3.new(12.75, 0, 0)).X,
   439→		zb = hit(Vector3.new(0, 0, -40), Vector3.new(cx, 0, -29.5)).Z,
   440→		zf = -2.5,
   441→	}
   442→	return room.box
   443→end

   445→-- one run of wainscoting on a wall face from a to b (floor height); n points into the room
   446→local function wainscot(parent, a, b, n, look)
   447→	local len = (b - a).Magnitude
   448→	if len < 1.5 then return end
   449→	local mid = (a + b) / 2
   450→	local base = CFrame.lookAt(mid, mid + n) -- -Z points into the room, X runs along the wall
   451→	local H = 3.8

   582→-- the hotel's rating (Hotel attribute HotelStars): the average of its rooms' stars, blended half and half with
   583→-- the guests' reviews once there are any. The last 20 reviews are published as JSON (Hotel attribute Reviews).
   584→local reviews = {} -- newest first: { name, stars, text, room }
   585→local reviewsJSON
   586→-- the saved reviews arrive on the hotel attribute when a save loads (Save)
   587→local function loadReviews()
   588→	local j = HOTEL:GetAttribute("Reviews")
   589→	if j == reviewsJSON or type(j) ~= "string" then return end
   590→	reviewsJSON = j
   591→	local ok, t = pcall(HttpService.JSONDecode, HttpService, j)
   592→	if ok and type(t) == "table" then reviews = t end
   593→end
   594→local function updateRating()
   595→	loadReviews()
   596→	local sum, n = 0, 0
   597→	for _, r in pairs(rooms) do
   598→		sum += r.model:GetAttribute("Stars") or 1
   599→		n += 1
   600→	end
   601→	local rating = n > 0 and sum / n or 1
   602→	if #reviews > 0 then
   603→		local rs = 0
   604→		local wn = 0
   605→		for _, r in ipairs(reviews) do rs += r.stars * (r.w or 1); wn += (r.w or 1) end -- influencers count more
   606→		local avg = rs / wn
   607→		HOTEL:SetAttribute("ReviewStars", math.floor(avg * 10 + 0.5) / 10)
   608→		rating = rating * 0.5 + avg * 0.5
   609→	end
   610→	HOTEL:SetAttribute("HotelStars", math.floor(rating * 10 + 0.5) / 10)
   611→end

  1049→-- the first side of the bed with room to stand
  1050→local function bedSide(room)
  1051→	local bed = core(room, "Bed")
  1052→	if not bed then return nil end
  1053→	local piv = bed:GetPivot()
  1054→	local zone = HOTEL.BuildZones:FindFirstChild(room.model.Name)
  1055→	sideParams.FilterDescendantsInstances = { HOTEL.Shell, HOTEL.Rooms, PLACED }
  1056→	local first
  1057→	for _, o in ipairs(Catalog.CORE.Bed.sides) do
  1058→		local p = piv * o
  1059→		first = first or p
  1060→		local inZone = zone and Build.zoneAt(p) == zone
  1061→		local clear = inZone
  1062→		if clear then
  1063→			for _, hit in ipairs(workspace:GetPartBoundsInBox(CFrame.new(p), Vector3.new(2, 3.6, 2), sideParams)) do
  1064→				if hit.CanCollide and not hit:IsDescendantOf(bed) then
  1065→					clear = false
  1066→					break
  1067→				end
  1068→			end
  1069→		end
  1070→		if clear then return p end
  1071→	end
  1072→	return first

  1687→local function faceTo(g, look)
  1688→	if not look then return end
  1689→	local p = g.root.Position
  1690→	local dir = flat(look - p)
  1691→	if dir.Magnitude < 0.05 then return end
  1692→	g.root.CFrame = CFrame.lookAt(p, p + dir)
  1693→end
  1694→
  1695→-- turn on the spot to face a direction over a moment (no snapping), eased; with `to`, also ease the last
  1696→-- stud or so onto that exact spot (the Humanoid stops short and twists about if made to walk that last bit)
  1697→local function turnTo(g, dir, to)
  1698→	dir = flat(dir)
  1699→	if dir.Magnitude < 0.05 or not g.alive or g.posed then return end
  1700→	local p0 = g.root.Position
  1701→	local slide = to and Vector3.new(to.X - p0.X, 0, to.Z - p0.Z) or Vector3.zero
  1702→	if slide.Magnitude > 2 then slide = slide.Unit * 2 end
  1703→	local look = g.root.CFrame.LookVector
  1704→	local a0 = math.atan2(-look.X, -look.Z)
  1705→	local a1 = math.atan2(-dir.X, -dir.Z)
  1706→	local d = (a1 - a0 + math.pi) % (2 * math.pi) - math.pi
  1707→	if math.abs(d) < math.rad(4) and slide.Magnitude < 0.1 then return end
  1708→	local secs = math.clamp(math.max(math.abs(d) / math.pi * 0.5, slide.Magnitude * 0.3), 0.15, 0.55)
  1709→	local t0 = os.clock()
  1710→	while g.alive and not g.posed do
  1711→		local k = math.min(1, (os.clock() - t0) / secs)
  1712→		local e = k * k * (3 - 2 * k)
  1713→		local y = g.root.Position.Y
  1714→		g.root.CFrame = CFrame.new(p0.X + slide.X * e, y, p0.Z + slide.Z * e) * CFrame.Angles(0, a0 + d * e, 0)
  1715→		if k >= 1 then break end
  1716→		task.wait()
  1717→	end
  1718→end

  1720→local function settle(g, target, timeout, tol)
  1721→	-- final approach: MoveTo decelerates into the exact spot, so the NPC glides to a
  1722→	-- stop instead of lurching past it (hum:Move steering never decelerates on its own)
  1723→	local t0 = os.clock()
  1724→	while g.alive and os.clock() - t0 < (timeout or 6) do
  1725→		local offset = flat(target - g.root.Position)
  1726→		if offset.Magnitude <= (tol or 1.2) then
  1727→			g.hum:Move(Vector3.zero)
  1728→			return true
  1729→		end
  1730→		g.hum:MoveTo(target)
  1731→		task.wait(0.1)
  1732→	end
  1733→	g.hum:Move(Vector3.zero)
  1734→	return flat(g.root.Position - target).Magnitude <= 2.5
  1735→end

  1738→-- on its path, which rounds corners smoothly, then glides into the last point. A watchdog nudges anyone who
  1739→-- stops making progress, and after three plans that fail the walker is simply put at its spot.

  1754→local LOOK = 2.4 -- studs ahead along the path that a walker aims at
  1755→
  1756→-- false if the walker is knocked off its path or keeps getting stuck (the caller plans again)
  1757→local SEP_R = 2.4 -- how far apart walkers try to stay from each other
  1758→local function follow(g, pts, tol)
  1759→	local n = #pts
  1760→	local seg = 1
  1761→	local lastPos, lastMove, nudges = g.root.Position, os.clock(), 0
  1762→	local sepAt, sepNear = 0, {} -- nearby walkers, rescanned a few times a second
  1763→	while g.alive and not g.posed do
  1764→		-- round 10e: someone's talking to them: stop, face them, and carry on once they're done
  1765→		if g.talkUntil and os.clock() < g.talkUntil then
  1766→			g.hum:Move(Vector3.zero)
  1767→			while g.alive and g.talkUntil and os.clock() < g.talkUntil do
  1768→				local pr = g.talkPlayer and g.talkPlayer.Character and g.talkPlayer.Character:FindFirstChild("HumanoidRootPart")
  1769→				if not pr or flat(pr.Position - g.root.Position).Magnitude > 16 then break end
  1770→				faceTo(g, pr.Position)
  1771→				task.wait(0.25)
  1772→			end
  1773→			g.talkUntil, g.talkPlayer = nil, nil
  1774→			lastPos, lastMove = g.root.Position, os.clock()
  1775→		end
  1776→		local pos = g.root.Position
  1777→		-- the closest point on the current leg; move on to the next leg once this one is passed
  1778→		local a, b, t
  1779→		while true do
  1780→			a, b = pts[seg], pts[seg + 1]
  1781→			local ab = flat(b - a)
  1782→			local len2 = ab:Dot(ab)
  1783→			t = len2 > 1e-4 and math.clamp(flat(pos - a):Dot(ab) / len2, 0, 1) or 1
  1784→			if seg < n - 1 and (t >= 0.999 or flat(b - pos).Magnitude < 1.2) then seg += 1 else break end
  1785→		end
  1786→		local proj = a + (b - a) * t
  1787→		if flat(pos - proj).Magnitude > 6 then g.hum:Move(Vector3.zero) return false end
  1788→		local last = pts[n]
  1789→		if seg == n - 1 and flat(last - pos).Magnitude < LOOK + 0.6 then
  1790→			return settle(g, last, 5, tol)
  1791→		end
  1792→		-- the carrot: LOOK studs further along the path from the closest point
  1793→		local left, from, k = LOOK, proj, seg
  1794→		local carrot = b
  1795→		while true do
  1796→			local to = pts[k + 1]
  1797→			local leg = flat(to - from)
  1798→			local d = leg.Magnitude
  1799→			if d >= left then carrot = from + leg.Unit * left break end
  1800→			left -= d
  1801→			if k + 1 >= n then carrot = to break end
  1802→			from, k = to, k + 1
  1803→		end
  1804→		local dir = flat(carrot - pos)
  1805→		-- round 16: steer apart from nearby walkers, so guests and staff spread out
  1806→		-- instead of walking through each other (off within 3 studs of the destination,
  1807→		-- so queue spots and poses still land exactly where they should)
  1808→		-- round 18: real avoidance. Each walker predicts where the people round it (guests, staff and players)
  1809→		-- will be over the next second and a half and sidesteps early, keeping right like people do; someone
  1810→		-- coming straight at them makes them ease off; a near miss that still touches is a bump (Voice.bump)
  1811→		if os.clock() - sepAt > 0.2 then
  1812→			sepAt = os.clock()
  1813→			table.clear(sepNear)
  1814→			for _, m in ipairs(GUESTS:GetChildren()) do
  1815→				local r = m:FindFirstChild("HumanoidRootPart")
  1816→				if r and r ~= g.root and flat(r.Position - pos).Magnitude < 12 then table.insert(sepNear, r) end
  1817→			end
  1818→			for _, pl in ipairs(Players:GetPlayers()) do
  1819→				local r = pl.Character and pl.Character:FindFirstChild("HumanoidRootPart")
  1820→				if r and flat(r.Position - pos).Magnitude < 12 then table.insert(sepNear, r) end
  1821→			end
  1822→		end
  1823→		local move = dir.Magnitude > 0.05 and dir.Unit or Vector3.zero
  1824→		local pace = 1
  1825→		if move.Magnitude > 0 and flat(pts[n] - pos).Magnitude > 2.5 then
  1826→			local myV = move * g.hum.WalkSpeed
  1827→			local right = Vector3.new(-move.Z, 0, move.X)
  1828→			local steer = Vector3.zero
  1829→			for _, r in ipairs(sepNear) do
  1830→				if r.Parent and math.abs(r.Position.Y - pos.Y) < 4 then
  1831→					local rel = flat(r.Position - pos)
  1832→					local d = rel.Magnitude
  1833→					if d > 0.01 and d < 10 then
  1834→						if d < SEP_R then steer -= rel.Unit * (1 - d / SEP_R) * 1.1 end
  1835→						local relV = flat(r.AssemblyLinearVelocity) - myV
  1836→						local vv = relV:Dot(relV)
  1837→						local tc = vv > 0.01 and math.clamp(-rel:Dot(relV) / vv, 0, 1.5) or 0
  1838→						local close = rel + relV * tc
  1839→						if tc > 0.05 and close.Magnitude < 2.4 then
  1840→							local side = close:Dot(right)
  1841→							local s = (math.abs(side) < 0.35) and -1 or (side > 0 and -1 or 1)
  1842→							steer += right * s * (1 - close.Magnitude / 2.4) * (1.7 - tc)
  1843→							if rel.Unit:Dot(move) > 0.75 and d < 4.5 then pace = math.min(pace, 0.55) end
  1844→						end
  1845→						if d < 1.7 and rel.Unit:Dot(move) > 0.35 and Life.Voice then
  1846→							Life.Voice.bump(g.model, r.Parent)
  1847→						end
  1848→					end
  1849→				end
  1850→			end
  1851→			if steer.Magnitude > 0.05 then move = (move + steer).Unit end
  1852→		end
  1853→		if move.Magnitude > 0.05 then g.hum:Move(move * pace) end
  1854→		-- watchdog
  1855→		if flat(pos - lastPos).Magnitude > 0.5 then
  1856→			lastPos, lastMove = pos, os.clock()
  1857→		elseif os.clock() - lastMove > 1 then
  1858→			nudges += 1
  1859→			if nudges > 3 or dir.Magnitude < 0.05 then g.hum:Move(Vector3.zero) return false end
  1860→			g.root.CFrame = g.root.CFrame + dir.Unit * math.min(1.5, dir.Magnitude)
  1861→			lastPos, lastMove = g.root.Position, os.clock()
  1862→		end
  1863→		task.wait(0.05)
  1864→	end
  1865→	g.hum:Move(Vector3.zero)
  1866→	return false
  1867→end

  1869→local function walkTo(g, target, tol)
  1870→	if typeof(target) == "Instance" then target = target.Position end
  1871→	if not g.alive then return end
  1872→	for _ = 1, 3 do
  1873→		if not g.alive or g.posed then return end
  1874→		local pts = grid:path(g.root.Position, target) or { g.root.Position, target }
  1875→		if follow(g, pts, tol) then return end
  1876→	end
  1877→	-- still not there: put them there rather than leave anyone stuck
  1878→	if g.alive and not g.posed and flat(g.root.Position - target).Magnitude > 3 then
  1879→		g.root.CFrame = CFrame.new(target) * g.root.CFrame.Rotation
  1880→	end
  1881→	g.hum:Move(Vector3.zero)
  1882→end

  1932→-- the line outside (round 10c): one straight file up the front path, everyone facing the door. A guest walks
  1933→-- straight into their place (no stop-and-shuffle), turns smoothly to the door, and steps back in if bumped.
  1934→local function queueGo(g, slot)
  1935→	if not (slot and g.alive) or g.room then return end
  1936→	local q1 = NAV.Queue1.Position
  1937→	local p = Vector3.new(O.X, q1.Y, q1.Z + (slot - 1) * C.QUEUE_GAP)
  1938→	local face = Vector3.new(0, 0, -1)
  1939→	local d = flat(p - g.root.Position).Magnitude
  1940→	if d > 1.3 then

  1942→		-- the last stretch: straight in
  1943→		local t1, last, lastT = os.clock(), g.root.Position, os.clock()
  1944→		while g.alive and not g.room and g.slot == slot and os.clock() - t1 < 8 do
  1945→			if flat(p - g.root.Position).Magnitude <= 1.3 then break end

  1948→				last, lastT = g.root.Position, os.clock()
  1949→			elseif os.clock() - lastT > 1.5 then
  1950→				walkTo(g, p, 1.3) -- something's in the way: path round it
  1951→				break
  1952→			end
  1953→			task.wait(0.1)
  1954→		end
  1955→		g.hum:Move(Vector3.zero)
  1956→	end

  1958→	if g.alive and not g.room and g.slot == slot then turnTo(g, face, p) end

  4831→	if rng:NextNumber() < chance then

  4834→	end
